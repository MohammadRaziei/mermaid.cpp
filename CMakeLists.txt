cmake_minimum_required(VERSION 3.19)

set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ version selection")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/cmake")
include(DynamicVersion)

extract_version_string(
    HEADER_FILE ${CMAKE_CURRENT_LIST_DIR}/include/mermaid/mermaid.h
    VERSION_PREFIX MERMAID_
    OUTPUT_VAR PROJECT_VERSION
)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

project(mermaid LANGUAGES CXX VERSION ${PROJECT_VERSION})
message(STATUS "Project: ${PROJECT_NAME}@v${PROJECT_VERSION}")

if(MSVC)
    add_compile_options(/EHsc)
    set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
if(CYGWIN OR UNIX)
    add_compile_definitions(_GNU_SOURCE)
endif()

file(GLOB SOURCES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/src/*.cpp)

# --- C++ core library (static) ---
# Install layout: a plain `cmake --install` uses the usual prefix layout.
set(_mermaid_lib_dest   lib)
set(_mermaid_inc_dest   include)
set(_mermaid_cmake_dest lib/cmake/mermaid)

add_library(mermaid STATIC ${SOURCES})
# The public target is `mermaid::mermaid` - the same name whether the
# library comes from FetchContent/add_subdirectory (this alias) or from an
# installed package (find_package(mermaid), exported below).
add_library(mermaid::mermaid ALIAS mermaid)

set_target_properties(mermaid PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON
    EXPORT_NAME mermaid
    OUTPUT_NAME mermaid
)
target_include_directories(mermaid
    PUBLIC
        $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${_mermaid_inc_dest}>
    PRIVATE
        ${PROJECT_SOURCE_DIR}/src)
if(NOT MSVC)
    target_compile_options(mermaid PRIVATE -Wall -Wextra -Wpedantic)
endif()

# --- install: find_package(mermaid CONFIG) -> mermaid::mermaid ---
include(CMakePackageConfigHelpers)
install(TARGETS mermaid EXPORT mermaidTargets
    ARCHIVE DESTINATION ${_mermaid_lib_dest}
    LIBRARY DESTINATION ${_mermaid_lib_dest})
install(DIRECTORY include/ DESTINATION ${_mermaid_inc_dest})
install(EXPORT mermaidTargets
    FILE mermaidTargets.cmake
    NAMESPACE mermaid::
    DESTINATION ${_mermaid_cmake_dest})

# mermaid_LIB_PATH / mermaid_INCLUDE_PATH for consumers that want plain
# paths. Produced with file(GENERATE) because the exact artifact name
# (libmermaid.a, mermaid.lib, ...) is only known to the generator.
file(RELATIVE_PATH _mermaid_rel_lib "/_/${_mermaid_cmake_dest}" "/_/${_mermaid_lib_dest}")
file(RELATIVE_PATH _mermaid_rel_inc "/_/${_mermaid_cmake_dest}" "/_/${_mermaid_inc_dest}")
string(REGEX REPLACE "/+$" "" _mermaid_rel_lib "${_mermaid_rel_lib}")
string(REGEX REPLACE "/+$" "" _mermaid_rel_inc "${_mermaid_rel_inc}")
file(GENERATE
    OUTPUT ${PROJECT_BINARY_DIR}/mermaidPaths.cmake
    CONTENT
"# Generated by mermaid's CMakeLists.txt - do not edit.
get_filename_component(mermaid_LIB_PATH
    \"\${CMAKE_CURRENT_LIST_DIR}/${_mermaid_rel_lib}/$<TARGET_FILE_NAME:mermaid>\" ABSOLUTE)
get_filename_component(mermaid_INCLUDE_PATH
    \"\${CMAKE_CURRENT_LIST_DIR}/${_mermaid_rel_inc}\" ABSOLUTE)
")

write_basic_package_version_file(
    ${PROJECT_BINARY_DIR}/mermaidConfigVersion.cmake
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion)
install(FILES
    ${PROJECT_SOURCE_DIR}/cmake/mermaidConfig.cmake
    ${PROJECT_BINARY_DIR}/mermaidPaths.cmake
    ${PROJECT_BINARY_DIR}/mermaidConfigVersion.cmake
    DESTINATION ${_mermaid_cmake_dest})

if(EXISTS ${PROJECT_SOURCE_DIR}/examples/main.cpp)
    add_executable(example examples/main.cpp)
    target_link_libraries(example PRIVATE mermaid::mermaid)
endif()

enable_testing()

# Only build tests when mermaid is the top-level project (see liburlparser
# for the rationale: embedded builds must not write into the source tree).
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
