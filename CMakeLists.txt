cmake_minimum_required(VERSION 3.19)

set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ version selection")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/cmake")
include(DynamicVersion)

extract_version_string(
    HEADER_FILE ${CMAKE_CURRENT_LIST_DIR}/include/mermaidcpp.h
    VERSION_PREFIX MERMAIDCPP_
    OUTPUT_VAR PROJECT_VERSION
)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

project(mermaidcpp LANGUAGES CXX VERSION ${PROJECT_VERSION})
message(STATUS "Project: ${PROJECT_NAME}@v${PROJECT_VERSION}")

if(MSVC)
    add_compile_options(/EHsc)
    set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
if(CYGWIN OR UNIX)
    add_compile_definitions(_GNU_SOURCE)
endif()

file(GLOB SOURCES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/src/*.cpp)

# --- C++ core library (static) ---
# Install layout: a plain `cmake --install` uses the usual prefix layout.
set(_mermaidcpp_lib_dest   lib)
set(_mermaidcpp_inc_dest   include)
set(_mermaidcpp_cmake_dest lib/cmake/mermaidcpp)

add_library(mermaidcpp STATIC ${SOURCES})
# The public target is `mermaidcpp::mermaidcpp` - the same name whether the
# library comes from FetchContent/add_subdirectory (this alias) or from an
# installed package (find_package(mermaidcpp), exported below).
add_library(mermaidcpp::mermaidcpp ALIAS mermaidcpp)

set_target_properties(mermaidcpp PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON
    EXPORT_NAME mermaidcpp
    OUTPUT_NAME mermaidcpp
)
target_include_directories(mermaidcpp
    PUBLIC
        $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${_mermaidcpp_inc_dest}>
    PRIVATE
        ${PROJECT_SOURCE_DIR}/src)
if(NOT MSVC)
    target_compile_options(mermaidcpp PRIVATE -Wall -Wextra -Wpedantic)
endif()

# --- install: find_package(mermaidcpp CONFIG) -> mermaidcpp::mermaidcpp ---
include(CMakePackageConfigHelpers)
install(TARGETS mermaidcpp EXPORT mermaidcppTargets
    ARCHIVE DESTINATION ${_mermaidcpp_lib_dest}
    LIBRARY DESTINATION ${_mermaidcpp_lib_dest})
install(DIRECTORY include/ DESTINATION ${_mermaidcpp_inc_dest})
install(EXPORT mermaidcppTargets
    FILE mermaidcppTargets.cmake
    NAMESPACE mermaidcpp::
    DESTINATION ${_mermaidcpp_cmake_dest})

# mermaidcpp_LIB_PATH / mermaidcpp_INCLUDE_PATH for consumers that want plain
# paths. Produced with file(GENERATE) because the exact artifact name
# (libmermaidcpp.a, mermaidcpp.lib, ...) is only known to the generator.
file(RELATIVE_PATH _mermaidcpp_rel_lib "/_/${_mermaidcpp_cmake_dest}" "/_/${_mermaidcpp_lib_dest}")
file(RELATIVE_PATH _mermaidcpp_rel_inc "/_/${_mermaidcpp_cmake_dest}" "/_/${_mermaidcpp_inc_dest}")
string(REGEX REPLACE "/+$" "" _mermaidcpp_rel_lib "${_mermaidcpp_rel_lib}")
string(REGEX REPLACE "/+$" "" _mermaidcpp_rel_inc "${_mermaidcpp_rel_inc}")
file(GENERATE
    OUTPUT ${PROJECT_BINARY_DIR}/mermaidcppPaths.cmake
    CONTENT
"# Generated by mermaidcpp's CMakeLists.txt - do not edit.
get_filename_component(mermaidcpp_LIB_PATH
    \"\${CMAKE_CURRENT_LIST_DIR}/${_mermaidcpp_rel_lib}/$<TARGET_FILE_NAME:mermaidcpp>\" ABSOLUTE)
get_filename_component(mermaidcpp_INCLUDE_PATH
    \"\${CMAKE_CURRENT_LIST_DIR}/${_mermaidcpp_rel_inc}\" ABSOLUTE)
")

write_basic_package_version_file(
    ${PROJECT_BINARY_DIR}/mermaidcppConfigVersion.cmake
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion)
install(FILES
    ${PROJECT_SOURCE_DIR}/cmake/mermaidcppConfig.cmake
    ${PROJECT_BINARY_DIR}/mermaidcppPaths.cmake
    ${PROJECT_BINARY_DIR}/mermaidcppConfigVersion.cmake
    DESTINATION ${_mermaidcpp_cmake_dest})

if(EXISTS ${PROJECT_SOURCE_DIR}/examples/main.cpp)
    add_executable(example examples/main.cpp)
    target_link_libraries(example PRIVATE mermaidcpp::mermaidcpp)
endif()

enable_testing()

# Only build tests when mermaidcpp is the top-level project (see liburlparser
# for the rationale: embedded builds must not write into the source tree).
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/tests)
endif()
